cmake_minimum_required(VERSION 3.20)
project(alarm_panel_desktop LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(PANEL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
set(LVGL_DIR "${PANEL_ROOT}/.pio/libdeps/esp32dev/lvgl" CACHE PATH "LVGL 8.4 source directory")
if(NOT EXISTS "${LVGL_DIR}/lvgl.h")
  message(FATAL_ERROR "LVGL missing. Run pio pkg install -e esp32dev, or set -DLVGL_DIR=/path/to/lvgl-8.4")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
file(GLOB_RECURSE LVGL_SOURCES CONFIGURE_DEPENDS "${LVGL_DIR}/src/*.c")
add_library(panel_lvgl STATIC ${LVGL_SOURCES})
target_include_directories(panel_lvgl PUBLIC "${LVGL_DIR}" "${PANEL_ROOT}/include")
target_compile_definitions(panel_lvgl PUBLIC LV_CONF_INCLUDE_SIMPLE=1 DESKTOP_SIM=1)
add_library(panel_workflow "${PANEL_ROOT}/src/workflow.cpp" "${PANEL_ROOT}/src/alarmo_events.cpp")
target_include_directories(panel_workflow PUBLIC "${PANEL_ROOT}/include")
add_executable(alarm-panel main.cpp "${PANEL_ROOT}/src/panel_font.c" "${PANEL_ROOT}/src/panel_clock_font.c" "${PANEL_ROOT}/src/ui.cpp" "${PANEL_ROOT}/src/alarm_state.cpp")
target_compile_definitions(alarm-panel PRIVATE DESKTOP_SIM=1)
target_link_libraries(alarm-panel PRIVATE panel_lvgl panel_workflow PkgConfig::SDL2)
target_compile_options(alarm-panel PRIVATE -Wall -Wextra)
add_executable(workflow-tests "${PANEL_ROOT}/tests/workflow_test.cpp")
target_link_libraries(workflow-tests PRIVATE panel_workflow)
enable_testing()
add_test(NAME workflows COMMAND workflow-tests)
add_test(NAME desktop-interactions COMMAND alarm-panel --smoke "${CMAKE_BINARY_DIR}/preview")
set_tests_properties(desktop-interactions PROPERTIES TIMEOUT 60 ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software")
